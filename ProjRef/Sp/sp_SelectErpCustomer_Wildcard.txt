Create PROCEDURE [dbo].[sp_SelectErpCustomer_Wildcard]
	@CompanyName as nvarchar(120), 
	@CompanyID as nvarchar(120), 
	@ErpDb as nvarchar(120), 
	@WebDb as nvarchar(120), 
	@QueryUserCode as nvarchar(120), 
	@CardType as nvarchar(1), 
	@wildCode as nvarchar(100)
AS

	declare @sqlstr as nvarchar(max)
	select @sqlStr = 'SELECT *
				,'+QUOTENAME(@CompanyName, '''') +' [CompanyName]
                ,'+QUOTENAME(@CompanyID, '''')+' [CompanyID]
                FROM ['+@ErpDb+'].[dbo].[OCRD] t0
                WHERE t0.CardCode LIKE '+ QUOTENAME( '%' +@wildCode + '%', '''') + '
				    or t0.CardName LIKE '+ QUOTENAME( '%' +@wildCode + '%', '''') + '
				    and t0.CardCode in (SELECT Distinct CardCode 
					FROM ['+@WebDb+'].[dbo].[USERCARD]
					WHERE UserCode = '+QUOTENAME(@QueryUserCode, '''')+' 
							AND CardType = '+QUOTENAME(@CardType, '''')+')'

--print @sqlstr
exec sp_executesql @sqlstr
