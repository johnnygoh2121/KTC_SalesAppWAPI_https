ALTER procedure [dbo].[sp_SelectGeoTrack] 
	@WebDb as nvarchar(120), 
	@QueryUserCode as nvarchar(120), 
	@QueryDate as date, 
	@Subsi as nvarchar(120)
	as

	declare @sqlstr nvarchar(max) 

	select @sqlstr = 'SELECT '+ QUOTENAME(@Subsi, '''') + '[Subsi] , * 
					FROM ['+@WebDb+'].[dbo].[FTAppGeoTrack] 
                    WHERE UserCode = '+QUOTENAME(@QueryUserCode, '''')+' 
                    AND Convert(date, ScheduleDate) = '+ QUOTENAME(convert(date,@QueryDate), '''')

--print @sqlstr
exec sp_executesql @sqlstr
