



use KTCW_COMMON
go 

alter procedure sp_SelectCog
@webDb as nvarchar(100), 
@cardCode as nvarchar(100), 
@startDt as datetime, 
@endDt as datetime, 
@subsi as nvarchar(100)

as 

declare @sql as nvarchar(max)

select @sql = '

select 
	'+QUOTENAME(@subsi, '''')+ ' [SubSi]
	,* 
	from ['+QUOTENAME(@webDb, '''')+'].[dbo].[COG]
where cardcode = ' +QUOTENAME(@cardCode, '''')+ ' 
and convert(date, docdate) >= '+QUOTENAME(@startDt, '''')+ '
and convert(date, docdate) <= '+QUOTENAME(@endDt, '''')


--print @sql
exec sp_executesql @sql
