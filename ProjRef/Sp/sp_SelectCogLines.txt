alter  procedure sp_SelectCogLines
@webDb as nvarchar(100), 
@subsi as nvarchar(100),
@docEntry as int 

as 

declare @sql as nvarchar(max)

select @sql = '
select '+QUOTENAME(@subsi, '''')+' [SubSi] 
		, * 
from ['+@webDb+'].[dbo].[COG1] t0
where t0.DocEntry = '+QUOTENAME(@docEntry, '''')+'

'

--print @sql
exec sp_executesql @sql
