

use KTCW_COMMON
go 

alter procedure sp_SelectCreditMemo_Approval 
@webDb as nvarchar(120), 
@docEntry as int

as

declare @sql as nvarchar(max)
select @sql = '
select t0.USERCODE, t1.USERNAME, t0.APPRDATE, t0.APPRREM [ApprRemark]
		from ['+@webDb+'].[dbo].[PAF4] t0 
		left join ['+@webDb+'].[dbo].[USERS] t1 on t0.USERCODE = t1.USERCODE 
where DOCENTRY = ' + QUOTENAME(@docEntry, '''')

exec sp_executesql @sql
